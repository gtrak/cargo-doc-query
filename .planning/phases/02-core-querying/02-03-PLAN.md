# Locked decisions from RESEARCH.md
- Automatic inference: Command infers whether the path refers to a type or trait
- Type queries: Returns methods and trait implementations for that type
- Trait queries: Returns the trait definition only (methods, associated types) — NOT implementing types
- Optional `--include=docs` flag includes doc comments in output
- Optional `--include=private` flag includes non-public items
- Multiple matches: Show all matches, do not prompt user for selection
- Fail fast: Query fails immediately on non-existent paths, ambiguous matches
</context>

```rust
impl CacheStore {

    /// Load the most recent index from cache (latest cache file)
    pub fn load_current(&self) -> Result<Option<SerializableIndex>> {
        let entries = std::fs::read_dir(&self.cache_dir)
            .context("Failed to read cache directory")?;

        let mut latest_mtime: Option<std::time::SystemTime> = None;
        let mut latest_path: Option<PathBuf> = None;

        for entry in entries {
            let entry = entry.context("Failed to read cache entry")?;
            let path = entry.path();

            if path.extension().and_then(|s| s.to_str()) == Some("idx") {
                let metadata = entry.metadata()
                    .context("Failed to read file metadata")?;

                let mtime = metadata.modified()
                    .context("Failed to get modification time")?;

                match &latest_mtime {
                    None => {
                        latest_mtime = Some(mtime);
                        latest_path = Some(path);
                    }
                    Some(current_latest) => {
                        if mtime > *current_latest {
                            latest_mtime = Some(mtime);
                            latest_path = Some(path);
                        }
                    }
                }
            }
        }

        match latest_path {
            Some(path) => {
                let data = std::fs::read(&path).context("Failed to read cache file")?;
                let index = from_bytes(&data).context("Failed to deserialize index")?;
                Ok(Some(index))
            }
            None => Ok(None),
        }
    }
}
```

```rust
use anyhow::{Context, Result};
use rustdoc_types::{Crate, Item, ItemEnum, Impl, Function, Type, Id};
use std::collections::HashMap;
use std::fs;
use crate::cache::store::SerializableIndex;
use crate::query::lookup::PathResolver;
use crate::query::format::TypeFormatter;
use crate::types::doc::DocExtractor;
use crate::types::query::*;

#[derive(Debug, Clone)]
pub struct QueryOptions {
    pub kind: QueryKind,
    pub include_docs: bool,
    pub include_private: bool,
}

#[derive(Debug, Clone, PartialEq)]
pub enum QueryKind {
    Methods,
    Traits,
    Types,
    All,
}

pub struct QueryEngine {
    index: SerializableIndex,
    crates: HashMap<String, Crate>,
}

impl QueryEngine {
    pub fn new(index: SerializableIndex) -> Self {
        Self {
            index,
            crates: HashMap::new(),
        }
    }

    /// Load a crate's rustdoc JSON into memory
    fn load_crate(&mut self, crate_name: &str, crate_version: &str) -> Result<&Crate> {
        // Check if already loaded
        let key = format!("{}::{}", crate_name, crate_version);
        if self.crates.contains_key(&key) {
            return Ok(&self.crates[&key]);
        }

        // Find the crate node
        let crate_node = self.index.nodes.iter()
            .find(|n| n.name == crate_name && n.version == crate_version)
            .ok_or_else(|| anyhow::anyhow!("Crate {} v{} not found in index", crate_name, crate_version))?;

        // Load rustdoc JSON
        let json_path = &crate_node.json_path;
        let json_str = fs::read_to_string(json_path)
            .with_context(|| format!("Failed to read rustdoc JSON from {}", json_path))?;

        let krate: Crate = serde_json::from_str(&json_str)
            .with_context(|| format!("Failed to parse rustdoc JSON from {}", json_path))?;

        self.crates.insert(key, krate);
        Ok(&self.crates[&key])
    }

    /// Execute a query
    pub fn query(&mut self, path: &str, options: &QueryOptions, crate_filter: Option<&str>) -> Result<QueryResponse> {
        let mut matches = Vec::new();

        // Find all crates containing matching items
        for crate_node in self.index.nodes.iter() {
            if let Some(filter) = crate_filter {
                if crate_node.name != filter {
                    continue;
                }
            }

            let krate = self.load_crate(&crate_node.name, &crate_node.version)?;

            let items = PathResolver::find_by_path(krate, path);

            if items.is_empty() {
                continue;
            }

            for (id, item) in items {
                let kind = Self::item_kind(item)?;
                let content = self.extract_content(krate, id, item, &kind, options)?;

                matches.push(QueryMatch {
                    crate_name: crate_node.name.clone(),
                    version: crate_node.version.clone(),
                    fully_qualified_path: self.get_qualified_path(krate, id)?,
                    kind,
                    content,
                });
            }
        }

        if matches.is_empty() {
            return Err(anyhow::anyhow!("No items found matching path: {}", path));
        }

        Ok(QueryResponse {
            query: path.to_string(),
            matches,
            errors: Vec::new(),
        })
    }

    /// Get the item kind for output
    fn item_kind(item: &Item) -> Result<String> {
        Ok(match &item.inner {
            ItemEnum::Struct(_) => "type",
            ItemEnum::Enum(_) => "type",
            ItemEnum::Union(_) => "type",
            ItemEnum::Trait(_) => "trait",
            ItemEnum::TraitAlias(_) => "trait",
            ItemEnum::TypeAlias(_) => "type",
            ItemEnum::Function(_) => "function",
            ItemEnum::Module(_) => "module",
            ItemEnum::Impl(_) => "impl",
            ItemEnum::Constant(_) => "constant",
            ItemEnum::Static(_) => "static",
            _ => "unknown",
        }.to_string())
    }

    /// Get fully qualified path for an item
    fn get_qualified_path(&self, krate: &Crate, id: Id) -> Result<String> {
        krate.paths.get(&id)
            .map(|summary| summary.path.clone())
            .ok_or_else(|| anyhow::anyhow!("No path found for item"))
    }

    /// Extract content based on item kind and query options
    fn extract_content(
        &self,
        krate: &Crate,
        id: Id,
        item: &Item,
        kind: &str,
        options: &QueryOptions,
    ) -> Result<QueryContent> {
        match kind {
            "type" => {
                Ok(QueryContent::Type(self.extract_type_result(krate, id, item, options)?))
            }
            "trait" => {
                Ok(QueryContent::Trait(self.extract_trait_result(krate, item, options)?))
            }
            _ => Err(anyhow::anyhow!("Unsupported item kind: {}", kind)),
        }
    }

    /// Extract type query result (methods + trait implementations)
    fn extract_type_result(
        &self,
        krate: &Crate,
        id: Id,
        item: &Item,
        options: &QueryOptions,
    ) -> Result<TypeResult> {
        // Determine type kind
        let kind_str = match &item.inner {
            ItemEnum::Struct(_) => "struct",
            ItemEnum::Enum(_) => "enum",
            ItemEnum::Union(_) => "union",
            ItemEnum::TypeAlias(_) => "type alias",
            _ => "type",
        };

        // Extract inherent methods (impl without trait)
        let mut methods = Vec::new();
        let mut trait_impls = Vec::new();

        for item_ref in krate.index.values() {
            if let ItemEnum::Impl(impl_block) = &item_ref.inner {
                // Check if this impl is for our type
                if self.impl_is_for_type(impl_block, id) {
                    if let Some(trait_path) = &impl_block.trait_ {
                        // Trait implementation
                        let trait_methods = self.extract_impl_methods(krate, impl_block, options)?;
                        trait_impls.push(TraitImplOutput {
                            trait_name: trait_path.path.clone(),
                            trait_path: trait_path.path.clone(),
                            methods: trait_methods,
                            provided_methods: Vec::new(),
                            generic_args: None,
                        });
                    } else {
                        // Inherent impl (methods)
                        let impl_methods = self.extract_impl_methods(krate, impl_block, options)?;
                        methods.extend(impl_methods);
                    }
                }
            }
        }

        Ok(TypeResult {
            kind: kind_str.to_string(),
            methods,
            trait_implementations: trait_impls,
        })
    }

    /// Check if impl block is for a specific type
    fn impl_is_for_type(&self, impl_block: &Impl, type_id: Id) -> bool {
        match &impl_block.for_ {
            Type::ResolvedPath(path) => path.id == type_id,
            _ => false,
        }
    }

    /// Extract methods from an impl block
    fn extract_impl_methods(
        &self,
        krate: &Crate,
        impl_block: &Impl,
        options: &QueryOptions,
    ) -> Result<Vec<MethodOutput>> {
        let mut methods = Vec::new();

        for item_id in &impl_block.items {
            if let Some(item) = krate.index.get(item_id) {
                if let ItemEnum::Function(func) = &item.inner {
                    methods.push(self.extract_method(item, func, options)?);
                }
            }
        }

        Ok(methods)
    }

    /// Extract trait query result (definition + methods + associated types)
    fn extract_trait_result(
        &self,
        krate: &Crate,
        item: &Item,
        options: &QueryOptions,
    ) -> Result<TraitResult> {
        if let ItemEnum::Trait(trait_def) = &item.inner {
            let trait_path = self.get_qualified_path(krate, item.id)?;

            // Extract trait methods
            let mut methods = Vec::new();
            for item_id in &trait_def.items {
                if let Some(func_item) = krate.index.get(item_id) {
                    if let ItemEnum::Function(func) = &func_item.inner {
                        methods.push(self.extract_method(func_item, func, options)?);
                    }
                }
            }

            // Extract associated types
            let mut associated_types = Vec::new();
            for item_id in &trait_def.items {
                if let Some(type_item) = krate.index.get(item_id) {
                    if let ItemEnum::TypeAlias(ty_alias) = &type_item.inner {
                        let bounds = ty_alias.generics.params
                            .iter()
                            .map(|p| p.name.clone())
                            .collect::<Vec<_>>()
                            .join(", ");

                        let default = ty_alias.type_
                            .as_ref()
                            .map(|ty| TypeFormatter::format_type(ty));

                        associated_types.push(AssociatedTypeOutput {
                            name: type_item.name.clone().unwrap_or_default(),
                            bounds: if bounds.is_empty() { None } else { Some(bounds) },
                            default,
                        });
                    }
                }
            }

            Ok(TraitResult {
                name: item.name.clone().unwrap_or_default(),
                path: trait_path,
                methods,
                associated_types,
                provided_methods: Vec::new(), // TODO: Extract provided methods
            })
        } else {
            Err(anyhow::anyhow!("Item is not a trait"))
        }
    }

    /// Extract a single method output
    fn extract_method(
        &self,
        item: &Item,
        func: &Function,
        options: &QueryOptions,
    ) -> Result<MethodOutput> {
        let signature = TypeFormatter::format_signature(&func.sig);
        let return_type = TypeFormatter::format_return_type(&func.sig.decl.output);
        let docs = if options.include_docs {
            DocExtractor::extract_docs(item)
        } else {
            None
        };

        Ok(MethodOutput {
            name: item.name.clone().unwrap_or_default(),
            signature,
            return_type,
            visibility: "public".to_string(), // TODO: Extract from rustdoc-types
            is_public: true,
            docs,
            is_trait_method: false, // TODO: Determine if trait method
        })
    }
}
```
