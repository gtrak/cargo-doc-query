- Retrieves full Item via Id from index HashMap (Tier 2)

Implementation:
```rust
use anyhow::Result;
use rustdoc_types::{Crate, Id, Item, ItemSummary};
use std::path::PathBuf;
use std::collections::HashMap;

pub struct PathResolver;

impl PathResolver {
    /// Find item by fully qualified path
    /// Returns all matching items (may be multiple across different crates)
    pub fn find_by_path(krate: &Crate, path: &str) -> Vec<(Id, &Item)> {
        krate.paths
            .iter()
            .filter(|(_, summary)| Self::path_matches(&summary.path, path))
            .filter_map(|(id, _)| krate.index.get(id).map(|item| (*id, item)))
            .collect()
    }

    /// Find items by path across all loaded crates
    pub fn find_by_path_in_crates(
        crates: &HashMap<String, Crate>,
        path: &str,
        crate_filter: Option<&str>,
    ) -> Vec<(String, Id, &Item)> {
        let mut matches = Vec::new();

        for (crate_name, krate) in crates {
            if let Some(filter) = crate_filter {
                if crate_name != filter {
                    continue;
                }
            }

            for (id, item) in Self::find_by_path(krate, path) {
                matches.push((crate_name.clone(), id, item));
            }
        }

        matches
    }

    /// Check if a path matches the query
    /// Support both exact match and suffix match (e.g., "Vec" matches "std::vec::Vec")
    fn path_matches(item_path: &str, query_path: &str) -> bool {
        if item_path == query_path {
            return true;
        }

        // Suffix match: query "Vec" matches "std::vec::Vec"
        if item_path.ends_with(&format!("::{}", query_path)) {
            return true;
        }

        false
    }

    /// Get item by Id from a specific crate
    pub fn get_item(krate: &Crate, id: Id) -> Option<&Item> {
        krate.index.get(&id)
    }
}
```

use rustdoc_types::{Type, FunctionSignature, FnDecl, FnReturnType};
use crate::types::doc::DocExtractor;

pub struct TypeFormatter;

impl TypeFormatter {
    /// Format a Type as a string for JSON output
    pub fn format_type(ty: &Type) -> String {
        match ty {
            Type::ResolvedPath(path) => path.path.clone(),
            Type::Generic(name) => name.clone(),
            Type::Primitive(prim) => prim.clone(),
            Type::Tuple(types) => {
                let inner: Vec<String> = types.iter().map(Self::format_type).collect();
                if inner.len() == 1 {
                    format!("({},)", inner[0])
                } else {
                    format!("({})", inner.join(", "))
                }
            }
            Type::Array(ty, len) => {
                format!["[{}; {}]", Self::format_type(ty), len]
            }
            Type::Slice(ty) => {
                format!["[{}]", Self::format_type(ty)]
            }
            Type::Reference { lifetime, mutable, elem } => {
                let lt = lifetime.as_ref().map(|l| format!("{} ", l)).unwrap_or_default();
                let mut_str = if *mutable { "mut " } else { "" };
                format!("&{}{}{}", lt, mut_str, Self::format_type(elem))
            }
            Type::RawPointer { mutable, constness, elem } => match (mutable, constness) {
                (true, false) => format!("*mut {}", Self::format_type(elem)),
                (false, true) => format!("*const {}", Self::format_type(elem)),
                _ => format!("*ptr {}", Self::format_type(elem)),
            }
            Type::BorrowedRef { lifetime, mutable, elem } => {
                let lt = lifetime.as_ref().map(|l| format!("{} ", l)).unwrap_or_default();
                let mut_str = if *mutable { "mut " } else { "" };
                format!("&{}{}{}", lt, mut_str, Self::format_type(elem))
            }
            Type::FunctionPointer { decl, generic_params, .. } => {
                let args: Vec<String> = decl.inputs
                    .iter()
                    .map(|arg| Self::format_type(arg))
                    .collect();

                let return_str = match &decl.output {
                    FnReturnType::Default => String::new(),
                    FnReturnType::Return(ty) => format!(" -> {}", Self::format_type(ty)),
                };

                let generics = if generic_params.is_empty() {
                    String::new()
                } else {
                    format!("<{}>", generic_params.join(", "))
                };

                format!("fn{}({})", generics, args.join(", ")) + &return_str
            }
            Type::Dynamic(bounds, _lifetime) => {
                format!("dyn {}", bounds.join(" + "))
            }
            Type::Unit => "()".to_string(),
            _ => "<complex type>".to_string(),
        }
    }

    /// Format a function signature for JSON output
    pub fn format_signature(sig: &FunctionSignature) -> String {
        let args: Vec<String> = sig.decl.inputs
            .iter()
            .map(|arg| Self::format_type(arg))
            .collect();

        let return_str = match &sig.decl.output {
            FnReturnType::Default => String::new(),
            FnReturnType::Return(ty) => format!(" -> {}", Self::format_type(ty)),
        };

        let generics = sig.generics.params
            .iter()
            .map(|p| p.name.clone())
            .collect::<Vec<_>>()
            .join(", ");

        let generics_str = if generics.is_empty() {
            String::new()
        } else {
            format!("<{}>", generics)
        };

        format!("fn{}({})", generics_str, args.join(", ")) + &return_str
    }

    /// Format return type for JSON output
    pub fn format_return_type(output: &FnReturnType) -> String {
        match output {
            FnReturnType::Default => "()".to_string(),
            FnReturnType::Return(ty) => Self::format_type(ty),
        }
    }
}
